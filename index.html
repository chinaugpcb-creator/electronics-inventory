  <script>

    let inventory = JSON.parse(localStorage.getItem('electronics_inventory') || '[]');

    let records = JSON.parse(localStorage.getItem('electronics_records') || '[]');


    function save() {

      localStorage.setItem('electronics_inventory', JSON.stringify(inventory));

      localStorage.setItem('electronics_records', JSON.stringify(records));

      updateStats();

      renderInventory();

      renderRecords();

      updateSelects();

    }


    function switchTab(tab) {

      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));

      document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));

      event.target.classList.add('active');

      document.getElementById(tab).classList.add('active');

    }


    function updateStats() {

      const totalValue = inventory.reduce((sum, item) => sum + (item.stock * (item.lastPrice || 0)), 0);

      const totalItems = inventory.length;

      const totalStock = inventory.reduce((sum, item) => sum + item.stock, 0);

      const lowStock = inventory.filter(item => item.stock <= item.minStock).length;


      document.getElementById('totalValue').textContent = '¥' + totalValue.toFixed(0);

      document.getElementById('totalItems').textContent = totalItems;

      document.getElementById('totalStock').textContent = totalStock;

      document.getElementById('lowStock').textContent = lowStock;

    }


    function renderInventory(filter = '') {

      const list = document.getElementById('inventoryList');

      const filtered = inventory.filter(item => 

        item.name.toLowerCase().includes(filter.toLowerCase()) ||

        item.model.toLowerCase().includes(filter.toLowerCase())

      );


      if (filtered.length === 0) {

        list.innerHTML = '<div class="empty"><div class="empty-icon">📦</div><div>暂无库存数据</div></div>';

        return;

      }


      list.innerHTML = filtered.map(item => {

        const stockStatus = item.stock <= item.minStock ? 'low' : item.stock <= item.minStock * 2 ? 'normal' : 'high';

        const badgeText = item.stock <= item.minStock ? '库存不足' : '库存正常';

        const badgeClass = `badge-${stockStatus}`;

        

        return `

          <div class="list-item">

            <div class="item-header">

              <div class="item-name">${item.name}</div>

              <div class="item-badge ${badgeClass}">${badgeText}</div>

            </div>

            <div style="font-size:13px; color:#64748b; margin-bottom:8px;">型号：${item.model} | 类别：${item.category}</div>

            <div class="item-details">

              <div class="detail-row">

                <span class="detail-label">当前库存</span>

                <span class="detail-value">${item.stock} ${item.unit}</span>

              </div>

              <div class="detail-row">

                <span class="detail-label">预警库存</span>

                <span class="detail-value">${item.minStock} ${item.unit}</span>

              </div>

              <div class="detail-row">

                <span class="detail-label">存放位置</span>

                <span class="detail-value">${item.location || '-'}</span>

              </div>

              <div class="detail-row">

                <span class="detail-label">库存价值</span>

                <span class="detail-value">¥${(item.stock * (item.lastPrice || 0)).toFixed(2)}</span>

              </div>

            </div>

            <div class="item-actions">

              <button class="btn btn-success" onclick="quickIn(${item.id})">入库</button>

              <button class="btn btn-warning" onclick="quickOut(${item.id})">出库</button>

              <button class="btn btn-danger" onclick="deleteItem(${item.id})">删除</button>

            </div>

          </div>

        `;

      }).join('');

    }


    function renderRecords() {

      const list = document.getElementById('recordsList');

      const sorted = [...records].sort((a, b) => b.time - a.time).slice(0, 20);


      if (sorted.length === 0) {

        list.innerHTML = '<div class="empty" style="padding:30px 0;"><div class="empty-icon">📜</div><div>暂无进销记录</div></div>';

        return;

      }


      list.innerHTML = sorted.map(record => `

        <div class="list-item">

          <div class="item-header">

            <span class="record-type type-${record.type === 'in' ? 'in' : 'out'}">${record.type === 'in' ? '入库' : '出库'}</span>

            <span style="font-size:12px; color:#94a3b8;">${new Date(record.time).toLocaleString()}</span>

          </div>

          <div style="font-size:14px; font-weight:600; color:#1e293b; margin:8px 0;">${record.itemName}</div>

          <div class="item-details">

            <div class="detail-row">

              <span class="detail-label">数量</span>

              <span class="detail-value">${record.quantity} ${record.unit}</span>

            </div>

            <div class="detail-row">

              <span class="detail-label">单价</span>

              <span class="detail-value">¥${record.price.toFixed(2)}</span>

            </div>

            <div class="detail-row">

              <span class="detail-label">${record.type === 'in' ? '供应商' : '客户'}</span>

              <span class="detail-value">${record.party}</span>

            </div>

            <div class="detail-row">

              <span class="detail-label">总额</span>

              <span class="detail-value" style="color:#2563eb;">¥${(record.quantity * record.price).toFixed(2)}</span>

            </div>

          </div>

          ${record.note ? `<div style="font-size:12px; color:#64748b; margin-top:8px;">备注：${record.note}</div>` : ''}

        </div>

      `).join('');

    }


    function updateSelects() {

      const inSelect = document.getElementById('inItemSelect');

      const outSelect = document.getElementById('outItemSelect');

      

      const options = inventory.map(item => 

        `<option value="${item.id}">${item.name} - ${item.model} (库存:${item.stock})</option>`

      ).join('');

      

      inSelect.innerHTML = '<option value="">请选择元器件</option>' + options;

      outSelect.innerHTML = '<option value="">请选择元器件</option>' + options;

    }


    function addItem() {

      const name = document.getElementById('itemName').value.trim();

      const model = document.getElementById('itemModel').value.trim();

      const category = document.getElementById('itemCategory').value;

      const unit = document.getElementById('itemUnit').value;

      const stock = parseInt(document.getElementById('itemStock').value) || 0;

      const minStock = parseInt(document.getElementById('itemMinStock').value) || 10;

      const location = document.getElementById('itemLocation').value.trim();


      if (!name || !model) {

        alert('请填写元器件名称和型号');

        return;

      }


      inventory.push({

        id: Date.now(),

        name, model, category, unit, stock, minStock, location,

        lastPrice: 0,

        createTime: Date.now()

      });


      document.getElementById('itemName').value = '';

      document.getElementById('itemModel').value = '';

      document.getElementById('itemLocation').value = '';

      document.getElementById('itemStock').value = '0';

      

      save();

      alert('添加成功！');

      switchTab('inventory');

    function showInForm() {

      document.getElementById('inForm').style.display = 'block';

      document.getElementById('outForm').style.display = 'none';

    }


    function showOutForm() {

      document.getElementById('inForm').style.display = 'none';

      document.getElementById('outForm').style.display = 'block';

    }


    function submitIn() {

      const itemId = parseInt(document.getElementById('inItemSelect').value);

      const quantity = parseInt(document.getElementById('inQuantity').value);

      const price = parseFloat(document.getElementById('inPrice').value);

      const supplier = document.getElementById('inSupplier').value.trim();

      const note = document.getElementById('inNote').value.trim();


      if (!itemId || !quantity || !price || !supplier) {

        alert('请填写完整信息');

        return;

      }


      const item = inventory.find(i => i.id === itemId);

      item.stock += quantity;

      item.lastPrice = price;


      records.push({

        id: Date.now(),

        type: 'in',

        itemId, itemName: item.name,

        quantity, price, unit: item.unit,

        party: supplier, note,

        time: Date.now()

      });


      document.getElementById('inQuantity').value = '';

      document.getElementById('inPrice').value = '';

      document.getElementById('inSupplier').value = '';

      document.getElementById('inNote').value = '';

      document.getElementById('inForm').style.display = 'none';


      save();

      alert('入库成功！');

    }


    function submitOut() {

      const itemId = parseInt(document.getElementById('outItemSelect').value);

      const quantity = parseInt(document.getElementById('outQuantity').value);

      const price = parseFloat(document.getElementById('outPrice').value);

      const customer = document.getElementById('outCustomer').value.trim();

      const note = document.getElementById('outNote').value.trim();


      if (!itemId || !quantity || !price || !customer) {

        alert('请填写完整信息');

        return;

      }


      const item = inventory.find(i => i.id === itemId);

      

      if (item.stock < quantity) {

        alert('库存不足！当前库存：' + item.stock);

        return;

      }


      item.stock -= quantity;


      records.push({

        id: Date.now(),

        type: 'out',

        itemId, itemName: item.name,

        quantity, price, unit: item.unit,

        party: customer, note,

        time: Date.now()

      });


      document.getElementById('outQuantity').value = '';

      document.getElementById('outPrice').value = '';

      document.getElementById('outCustomer').value = '';

      document.getElementById('outNote').value = '';

      document.getElementById('outForm').style.display = 'none';


      save();

      alert('出库成功！');

    }

    function deleteItem(id) {

      if (confirm('确定删除这个元器件吗？相关的进销记录也会被删除。')) {

        inventory = inventory.filter(item => item.id !== id);

        records = records.filter(record => record.itemId !== id);

        save();

      }

    }
